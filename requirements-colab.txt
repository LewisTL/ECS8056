# OpenVLA inference-only requirements for Google Colab.
#
# IMPORTANT: do NOT reinstall torch on Colab. Colab ships a torch build that is
# matched to its CUDA driver; overriding it tends to break GPU support. We pin
# only the libraries that govern OpenVLA model loading, plus quantisation deps.
#
# The trio below is the OpenVLA authors' known-good set. Newer transformers/timm/
# tokenizers cause the misleading "requires ... prismatic" load error.
transformers==4.40.1
tokenizers==0.19.1
timm==0.9.10

# huggingface_hub must stay compatible with transformers 4.40.1 (newer hub
# removes symbols that 4.40 still imports).
huggingface_hub==0.23.4

# Quantisation + device placement
accelerate==0.30.1
bitsandbytes>=0.45.0

# Required by the Llama/SentencePiece tokenizer conversion under transformers 4.40.1.
protobuf>=6.31.1

# Vision / arrays (usually already present on Colab, pinned loosely for safety)
pillow>=10.0.0
numpy>=1.24,<2.0

# Automated duplicate-target proposal (detect_duplicates.py). The OWLv2
# open-vocabulary detector ships with the transformers pin above, so no extra
# install is required for the detector itself. Target-noun extraction prefers a
# regex heuristic tuned to the short imperative instructions and falls back to
# spaCy only when the heuristic finds nothing, so spaCy is optional. To enable
# the fallback, install it and its small English model:
#   pip install spacy && python -m spacy download en_core_web_sm
